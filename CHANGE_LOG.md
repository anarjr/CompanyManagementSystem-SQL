#### Repository/Repositories/BaseRepository.cs
**Changes:** Complete refactor (class definition to DbContext pattern)
- Reason: Switch from List<T> to EF Core DbContext

**Before:**
```csharp

{

	public BaseRepository(List<T> datas) { _datas = datas; }

	public void Delete(T entity) { _datas.Remove(entity); }

	// ... more methods

```

**After:**
```csharp
public class BaseRepository<T> : IBaseRepository<T> where T : BaseEntity
{
	protected readonly AppDbContext _context;

	public BaseRepository(AppDbContext context) { _context = context; }

	public void Create(T entity)
	{
		_context.Set<T>().Add(entity);
		_context.SaveChanges();
	}

	public void Delete(T entity)
	{
		_context.Set<T>().Remove(entity);
		_context.SaveChanges();
	}

	public List<T> GetAll() { return _context.Set<T>().ToList(); }
	public T GetById(int id) { return _context.Set<T>().FirstOrDefault(m => m.Id == id); }
	// ... more methods
}
```
